function initializeSystem() {
  const ss = getSpreadsheet_();
  ensureAppOwner_();
  ensureSheet_(ss, "Readings");
  ensureSheet_(ss, "Daily Summary");
  ensureSheet_(ss, "Billing Summary");
  ensureSheet_(ss, "Estimated Bill");
  ensureTariffSettingsSheet_(ss);
  ensureMeterSettingsSheet_(ss);
  ensureEstimationProfileSheet_(ss);
  ensureNotifySheet_(ss);
  ensureNotesSheet_(ss);
  ensureGeneralSettingsSheet_(ss);
  ensureBillingHistorySheet_(ss);
  ensureReadingCollectionsSheet_(ss);
  removeObsoleteTriggers_();
  ensureFormTrigger_(ss);
  rebuildData();

}
function ensureAppOwner_(){
  const props=PropertiesService.getScriptProperties();
  const key="APP_OWNER_EMAIL";
  if(props.getProperty(key)) return;
  let email="";
  try{ email=String(Session.getEffectiveUser().getEmail()||"").trim().toLowerCase(); }catch(e){}
  if(email) props.setProperty(key,email);
}
function isAppOwner_(){
  const owner=String(PropertiesService.getScriptProperties().getProperty("APP_OWNER_EMAIL")||"").trim().toLowerCase();
  let email="";
  try{ email=String(Session.getActiveUser().getEmail()||"").trim().toLowerCase(); }catch(e){}
  if(!owner || !email) return false;
  return email===owner;
}
function requireAppOwner_(){
  if(!isAppOwner_()) throw new Error("Only the app owner can edit or delete reading records.");
}
function removeObsoleteTriggers_() {
  const obsolete = [
    "upgradeTwoMeterSystem"
  ];
  ScriptApp.getProjectTriggers().forEach(function(t) {
    if (obsolete.indexOf(t.getHandlerFunction()) !== -1) {
      ScriptApp.deleteTrigger(t);
    }
  });
}

function ensureFormTrigger_(ss) {
  const exists = ScriptApp.getProjectTriggers().some(function(t) {
    return t.getHandlerFunction() === "processFormSubmission";
  });
  if (!exists) {
    ScriptApp.newTrigger("processFormSubmission")
      .forSpreadsheet(ss)
      .onFormSubmit()

function installReminderTrigger() {
  const exists = ScriptApp.getProjectTriggers().some(function(t) {
    return t.getHandlerFunction() === "sendMissingReadingReminders";
  });

function rebuildData() {
  const ss = getSpreadsheet_();
  const responseSheet = findResponseSheet_(ss);
  if (!responseSheet) return;
  const readings = readFormResponses_(responseSheet);
  readings.sort(function(a, b) {
    return a.datetime - b.datetime;
  });
  applyCarryForward_(readings);
  rebuildReadingsSheet_(ss, readings);
  rebuildDailySummary_(ss, readings);
  rebuildBillingSummary_(ss, readings);
  rebuildEstimatedBillSheet_(ss, readings);
}
function readFormResponses_(sheet) {
  const values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  const headers = values[0];
  const dateIndex = findHeaderIndex_(headers, ["Reading Date"]);
  const timeIndex = findHeaderIndex_(headers, ["Reading Time"]);
  const newMeterIndex = findHeaderIndex_(headers, [
    "New Meter Reading (kWh)",
    "Meter Reading (kWh)"
  ]);
  const oldMeterIndex = findHeaderIndex_(headers, [
    "Old Meter Reading (kWh)"
  ]);
  const notesIndex = findHeaderIndex_(headers, ["Notes"]);
  if (dateIndex < 0 || timeIndex < 0) {
    throw new Error("Reading Date or Reading Time column was not found.");
  }
  const result = [];
  for (let i = 1; i < values.length; i++) {
    const row = values[i];
    if (row[dateIndex] === "" || row[timeIndex] === "") continue;
    const datetime = combineDateTime_(row[dateIndex], row[timeIndex]);
    if (!datetime) continue;
    let newReading = null;
    let oldReading = null;
    if (newMeterIndex >= 0) {
      const v = row[newMeterIndex];
      if (v !== "" && !isNaN(Number(v))) newReading = Number(v);
    }
    if (oldMeterIndex >= 0) {
      const v = row[oldMeterIndex];
      if (v !== "" && !isNaN(Number(v))) oldReading = Number(v);
    }
    result.push({
      datetime: datetime,
      newInput: newReading,
      oldInput: oldReading,
      newReading: null,
      oldReading: null,
      notes: notesIndex >= 0 ? row[notesIndex] : ""
    });
  }
  return result;
}
function applyCarryForward_(readings) {
  let lastNew = null;
  let lastOld = null;
  readings.forEach(function(r) {
    if (r.newInput !== null) lastNew = r.newInput;
    if (r.oldInput !== null) lastOld = r.oldInput;
    r.newReading = lastNew;
    r.oldReading = lastOld;
  });
}
function flagRawDecrease_(readings) {
  let lastNewInput = null;
  let lastOldInput = null;
  return readings.map(function(r) {
    let flag = "";
    if (r.newInput !== null) {
      if (lastNewInput !== null && r.newInput < lastNewInput) {
        flag += "Meter 1 reading decreased (check for a typo or meter change); ";
      }
      lastNewInput = r.newInput;
    }
    if (r.oldInput !== null) {
      if (lastOldInput !== null && r.oldInput < lastOldInput) {
        flag += "Meter 2 reading decreased (check for a typo or meter change); ";
      }
      lastOldInput = r.oldInput;
    }
    return flag;
  });
}
function rebuildReadingsSheet_(ss, readings) {
  const sheet = ensureSheet_(ss, "Readings");
  sheet.clearContents();
  sheet.getRange(1, 1, 1, 14).setValues([[
    "Date",
    "Time",
    "Date & Time",
    "New Meter Reading (kWh)",
    "Old Meter Reading (kWh)",
    "New Meter Difference (kWh)",
    "Old Meter Difference (kWh)",
    "Total Difference (kWh)",
    "5 PM Day",
    "Period",
    "Billing Month",
    "New Meter Status",
    "Old Meter Status",
    "Anomaly"
  ]]);
  styleHeader_(sheet, 14);
  if (!readings.length) return;
  const anomalyFlags = flagRawDecrease_(readings);
  const output = [];
  for (let i = 0; i < readings.length; i++) {
    const r = readings[i];
    let newDiff = "";
    let oldDiff = "";
    if (i > 0) {
      const previous = readings[i - 1];
      if (r.newReading !== null && previous.newReading !== null) {
        newDiff = r.newReading - previous.newReading;
      }
      if (r.oldReading !== null && previous.oldReading !== null) {
        oldDiff = r.oldReading - previous.oldReading;
      }
    }
    let totalDiff = "";
    if (newDiff !== "" || oldDiff !== "") {
      totalDiff =
        (newDiff === "" ? 0 : newDiff) +
        (oldDiff === "" ? 0 : oldDiff);
    }
    output.push([
      r.datetime,
      r.datetime,
      r.datetime,
      r.newReading === null ? "" : r.newReading,
      r.oldReading === null ? "" : r.oldReading,
      newDiff,
      oldDiff,
      totalDiff,
      formatFivePmDay_(r.datetime),
      isSolarPeriod_(r.datetime) ? "Solar/Daytime" : "Grid Period",
      getBillingMonth_(r.datetime),
      r.newInput === null ? "Unchanged/blank" : "Entered",
      r.oldInput === null ? "Unchanged/blank" : "Entered",
      anomalyFlags[i]
    ]);
  }
  sheet.getRange(2, 1, output.length, 14).setValues(output);
  sheet.getRange(2, 1, output.length, 1).setNumberFormat("dd/mm/yyyy");
  sheet.getRange(2, 2, output.length, 2).setNumberFormat("hh:mm AM/PM");
  sheet.getRange(2, 4, output.length, 5).setNumberFormat("0.00");
  sheet.autoResizeColumns(1, 14);
  sheet.setFrozenRows(1);
}
function rebuildDailySummary_(ss, readings) {
  const sheet = ensureSheet_(ss, "Daily Summary");
  sheet.clearContents();
  sheet.getRange(1, 1, 1, 9).setValues([[
    "5 PM Day",
    "Period Start",
    "Period End",
    "New Meter Consumption (kWh)",
    "Old Meter Consumption (kWh)",
    "Total Consumption (kWh)",
    "New Meter Reading at End",
    "Old Meter Reading at End",
    "Data Status"
  ]]);
  styleHeader_(sheet, 9);
  if (!readings.length) return;
  let day = getFivePmDayStart_(readings[0].datetime);
  const lastDay = getFivePmDayStart_(readings[readings.length - 1].datetime);
  const now = new Date();
  const output = [];
  while (day <= lastDay) {
    const start = new Date(day);
    const end = new Date(day);
    end.setDate(end.getDate() + 1);
    const startR = interpolatedReadingsAt_(readings, start);
    const endR = now >= end
      ? interpolatedReadingsAt_(readings, end)
      : readingsAtOrBefore_(readings, now);
    const newConsumption =
      calculateDifference_(startR.newReading, endR.newReading);
    const oldConsumption =
      calculateDifference_(startR.oldReading, endR.oldReading);
    const total = sumKnown_(newConsumption, oldConsumption);
    let status = "Complete";
    if (newConsumption === "" || oldConsumption === "") {
      status = "One meter has no baseline reading yet";
    }
    output.push([
      new Date(day),
      start,
      end,
      newConsumption,
      oldConsumption,
      total,
      endR.newReading === null ? "" : endR.newReading,
      endR.oldReading === null ? "" : endR.oldReading,
      status
    ]);
    day.setDate(day.getDate() + 1);
  }
  sheet.getRange(2, 1, output.length, 9).setValues(output);
  sheet.getRange(2, 1, output.length, 3)
    .setNumberFormat("dd/mm/yyyy hh:mm AM/PM");
  sheet.getRange(2, 4, output.length, 5).setNumberFormat("0.00");
  sheet.autoResizeColumns(1, 9);
  sheet.setFrozenRows(1);
}
function rebuildBillingSummary_(ss, readings) {
  const sheet = ensureSheet_(ss, "Billing Summary");
  sheet.clearContents();
  sheet.getRange(1, 1, 1, 13).setValues([[
    "Billing Month",
    "Period Start",
    "Period End",
    "Meter 1 Total (kWh)",
    "Meter 2 Total (kWh)",
    "Combined Total (kWh)",
    "Elapsed Days",
    "Meter 1 Daily Average",
    "Meter 2 Daily Average",
    "Combined Daily Average",
    "Meter 1 Projected",
    "Meter 2 Projected",
    "Combined Projected"
  ]]);
  styleHeader_(sheet, 13);
  if (!readings.length) return;
  const periods = {};
  readings.forEach(function(r) {
    const start = getBillingPeriodStart_(r.datetime);
    periods[start.getTime()] = {
      start: start,
      end: addMonth_(start)
    };
  });
  const keys = Object.keys(periods)
    .map(Number)
    .sort(function(a, b) { return a - b; });
  const output = [];
  keys.forEach(function(key) {
    const p = periods[key];
    const startR = interpolatedReadingsAt_(readings, p.start);
    const endR = interpolatedReadingsAt_(readings, p.end);
    const meter1 = calculateDifference_(startR.newReading, endR.newReading);
    const meter2 = calculateDifference_(startR.oldReading, endR.oldReading);
    const combined = sumKnown_(meter1, meter2);
    const periodReadings = readings.filter(function(r) {
      return r.datetime >= p.start && r.datetime < p.end;
    });
    const meterChange = estimateMeterChangeEvent_(readings, p.start, p.end);
    const meter1Days = meterOperatingDays_(readings, p.start, p.end, "new", meterChange);
    const meter2Days = meterOperatingDays_(readings, p.start, p.end, "old", meterChange);
    const avg1 = meter1Days > 0 && meter1 !== "" ? meter1 / meter1Days : "";
    const avg2 = meter2Days > 0 && meter2 !== "" ? meter2 / meter2Days : "";
    let elapsedDays = "";
    if (periodReadings.length >= 2) {
      elapsedDays =
        (periodReadings[periodReadings.length - 1].datetime -
         periodReadings[0].datetime) / 86400000;
    }
    const avgTotal = elapsedDays > 0 && combined !== "" ? combined / elapsedDays : "";
    const fullDays = (p.end - p.start) / 86400000;
    output.push([
      formatBillingMonth_(p.end),
      p.start,
      p.end,
      meter1,
      meter2,
      combined,
      elapsedDays,
      avg1,
      avg2,
      avgTotal,
      avg1 === "" ? "" : avg1 * fullDays,
      avg2 === "" ? "" : avg2 * fullDays,
      avgTotal === "" ? "" : avgTotal * fullDays
    ]);
  });
  if (!output.length) return;
  sheet.getRange(2, 1, output.length, 13).setValues(output);
  sheet.getRange(2, 2, output.length, 2)
    .setNumberFormat("dd/mm/yyyy hh:mm AM/PM");
  sheet.getRange(2, 4, output.length, 10).setNumberFormat("0.00");
  sheet.autoResizeColumns(1, 13);
  sheet.setFrozenRows(1);
}

  }
  if (sheet.getLastRow() === 0) {
    writeDefaultTariffSettings_(sheet);
  } else {
    const values = sheet.getDataRange().getValues();
    const requestedDefaults = {
      "0-100 Rate": 37.50,
      "100-200 Rate": 43.50
    };
    const oldDefaults = {
      "0-100 Rate": [22.44],
      "100-200 Rate": [28.91],
      "1-100 Rate": [22.44],
      "101-200 Rate": [28.91]
    };
    for (let i = 1; i < values.length; i++) {
      const key = String(values[i][0] || '').trim();
      if ((key === "1-100 Rate" || key === "101-200 Rate") &&
          oldDefaults[key].indexOf(Number(values[i][1])) !== -1) {
        const newKey = key === "1-100 Rate" ? "0-100 Rate" : "100-200 Rate";
        sheet.getRange(i + 1, 1).setValue(newKey);

          key === "1-100 Rate" ? 37.50 : 43.50

                 oldDefaults[key].indexOf(Number(values[i][1])) !== -1) {
        sheet.getRange(i + 1, 2).setValue(requestedDefaults[key]);
      }
    }
  }
  return sheet;
}

function calculateMeterBill_(units, tariff) {
  units = Number(units || 0);
  if (units < 0) units = 0;
  if (units === 0) {
    return {
      units: 0, slab: "No consumption", rate: 0, energy: 0, fixed: 0,
      adjustments: 0, duty: 0, gst: 0, tv: 0, otherFixed: 0, total: 0,
      steps: [{label:"Consumption", formula:"0 kWh", amount:0}]
    };
  }
  const activeSlabs = tariff.consumerType === "Protected" ? tariff.protectedSlabs : tariff.slabs;
  if (!activeSlabs || !activeSlabs.length) throw new Error("No tariff slabs configured for " + tariff.consumerType + ".");
  let slab = activeSlabs[activeSlabs.length - 1];
  for (let i = 0; i < activeSlabs.length; i++) {
    if (units >= activeSlabs[i].min && units <= activeSlabs[i].max) {
      slab = activeSlabs[i];
      break;
    }
  }
  const steps = [];
  let energy = 0;
  if (tariff.slabMode === "PROGRESSIVE") {
    let remaining = units;
    activeSlabs.forEach(function(s) {
      if (remaining <= 0) return;
      const width = s.max === Infinity
        ? remaining
        : Math.min(remaining, Math.max(0, s.max - s.min + 1));
      if (width <= 0) return;
      const amount = width * s.rate;
      energy += amount;
      steps.push({
        label: "Energy — " + slabLabel_(s),
        formula: fmtUnits_(width) + " × Rs " + fmtMoney_(s.rate) + " = Rs " + fmtMoney_(amount),
        amount: amount
      });
      remaining -= width;
    });
  } else {
    energy = units * slab.rate;
    steps.push({
      label: "Energy charges — applicable slab " + slabLabel_(slab),
      formula: fmtUnits_(units) + " × Rs " + fmtMoney_(slab.rate) + " = Rs " + fmtMoney_(energy),
      amount: energy
    });
  }
  const fixed = slab.fixed;
  steps.push({
    label: "Fixed charge",
    formula: "Applicable slab fixed charge = Rs " + fmtMoney_(fixed),
    amount: fixed
  });
  const perUnitAdjustments =
    tariff.adjustments.FCA + tariff.adjustments.QTA +
    tariff.adjustments.FC + tariff.adjustments.NJ;
  const adjustmentAmount = units * perUnitAdjustments;
  if (perUnitAdjustments !== 0) {
    steps.push({
      label: "Per-unit adjustments",
      formula: fmtUnits_(units) + " × Rs " + fmtMoney_(perUnitAdjustments) +
        " = Rs " + fmtMoney_(adjustmentAmount),
      amount: adjustmentAmount
    });
  }
  const otherFixed = tariff.adjustments.OtherFixed;
  if (otherFixed !== 0) {
    steps.push({
      label: "Other fixed",
      formula: "Rs " + fmtMoney_(otherFixed),
      amount: otherFixed
    });
  }
  const taxableBase = energy + fixed + adjustmentAmount + otherFixed;
  const duty = taxableBase * tariff.adjustments.ED / 100;
  const gst = (taxableBase + duty) * tariff.adjustments.GST / 100;
  const tv = tariff.adjustments.TV;
  if (tariff.adjustments.ED !== 0) {
    steps.push({
      label: "Electricity duty",
      formula: "(" + fmtMoney_(taxableBase) + ") × " + fmtMoney_(tariff.adjustments.ED) +
        "% = Rs " + fmtMoney_(duty),
      amount: duty
    });
  }
  if (tariff.adjustments.GST !== 0) {
    steps.push({
      label: "GST",
      formula: "(" + fmtMoney_(taxableBase + duty) + ") × " + fmtMoney_(tariff.adjustments.GST) +
        "% = Rs " + fmtMoney_(gst),
      amount: gst
    });
  }
  if (tv !== 0) {
    steps.push({
      label: "TV fee",
      formula: "Rs " + fmtMoney_(tv),
      amount: tv
    });
  }
  const total = energy + fixed + adjustmentAmount + duty + gst + tv + otherFixed;
  steps.push({
    label: "Estimated bill",
    formula: "All charges above = Rs " + fmtMoney_(total),
    amount: total,
    total: true
  });
  return {
    units: units,
    slab: slabLabel_(slab),
    rate: slab.rate,
    energy: energy,
    fixed: fixed,
    adjustments: adjustmentAmount,
    duty: duty,
    gst: gst,
    tv: tv,
    otherFixed: otherFixed,
    total: total,
    steps: steps
  };
}

function estimateMeterChangeEvent_(readings, periodStart, periodEnd) {
  const candidates=readings.filter(function(r){
    return r.datetime>=periodStart && r.datetime<periodEnd &&
      r.newInput!==null && r.oldInput!==null;
  });
  if(!candidates.length) return null;
  let event=candidates[candidates.length-1];
  let previousSameDay=null;
  const eventDayStart=getFivePmDayStart_(event.datetime);
  for(let i=0;i<readings.length;i++){
    const r=readings[i];
    if(r.datetime>=eventDayStart && r.datetime<event.datetime) {
      if(r.newInput!==null || r.oldInput!==null) previousSameDay=r;
    }
  }
  const dayStart=new Date(eventDayStart);
  const dayEnd=new Date(dayStart);
  dayEnd.setDate(dayEnd.getDate()+1);
  let estimated;
  if(previousSameDay) {
    estimated=new Date(
      previousSameDay.datetime.getTime()+
      (event.datetime.getTime()-previousSameDay.datetime.getTime())/2
    );
  } else {
    estimated=new Date(
      dayStart.getTime()+
      (event.datetime.getTime()-dayStart.getTime())/2
    );
  }
  estimated=new Date(Math.max(
    dayStart.getTime(),
    Math.min(dayEnd.getTime(),estimated.getTime())
  ));
  const dayProgress=(estimated-dayStart)/(dayEnd-dayStart)*100;
  const monthProgress=(estimated-periodStart)/(periodEnd-periodStart)*100;
  let activeAfter = "new";
  for (let i = 0; i < readings.length; i++) {
    const r = readings[i];
    if (r.datetime > event.datetime) {
      if (r.newInput !== null && r.oldInput === null) {
        activeAfter = "new";
        break;
      }
      if (r.oldInput !== null && r.newInput === null) {
        activeAfter = "old";
        break;
      }
    }
  }
  return {
    observedAt:event.datetime,
    estimatedAt:estimated,
    dayStart:dayStart,
    dayProgress:Math.max(0,Math.min(100,dayProgress)),
    monthProgress:Math.max(0,Math.min(100,monthProgress)),
    activeAfter:activeAfter
  };
}
function calculateProRata_(baseline,present,extendedDays,standardDays){
  if(baseline===null || present===null || baseline==="" || present==="") return null;
  const actual=Number(present)-Number(baseline);
  // Extended days are rounded to a whole day for billing math (e.g. 32.80 → 33).
  const days=Math.round(Number(extendedDays||0));
  const standard=Math.round(Number(standardDays||0));
  if(!isFinite(actual) || !isFinite(days) || !isFinite(standard) || days<=0 || standard<=0 || actual<0) return null;
  const daily=actual/days;
  const billed=Math.floor(daily*standard);
  return {
    baseline:Number(baseline),present:Number(present),extendedDays:days,
    actualUnits:actual,dailyAverage:daily,standardDays:standard,
    billedUnits:billed,adjustedPresent:Number(baseline)+billed,
    carryForward:actual-billed
  };
}

/*
 * Length in days of the billing period that *ends* at periodEnd
 * (same clock time one month earlier → periodEnd). Varies 28–31.
 */
function billingPeriodLengthDays_(periodEnd) {
  if (!periodEnd || isNaN(periodEnd.getTime())) return 30;
  const start = new Date(
    periodEnd.getFullYear(),
    periodEnd.getMonth() - 1,
    periodEnd.getDate(),
    periodEnd.getHours(),
    periodEnd.getMinutes(),
    periodEnd.getSeconds(),
    periodEnd.getMilliseconds()
  );
  const days = (periodEnd.getTime() - start.getTime()) / 86400000;
  return days > 0 ? days : 30;
}

/*
 * Map a collection month label ("Sep 2026") to the billing-period end
 * boundary (billing day/hour of that month).
 */
function periodEndFromMonthLabel_(label) {
  const raw = String(label || "").trim();
  const m = raw.match(/^([A-Za-z]{3})\s+(\d{4})$/);
  if (!m) return null;
  const map = {jan:0,feb:1,mar:2,apr:3,may:4,jun:5,jul:6,aug:7,sep:8,oct:9,nov:10,dec:11};
  const mi = map[m[1].toLowerCase()];
  if (mi === undefined) return null;
  const year = Number(m[2]);
  const gs = getGeneralSettings_();
  return new Date(year, mi, gs.billingDay, gs.billingHour, gs.billingMinute, 0, 0);
}

/*
 * Standard (billable) days for a collection: calendar length of the
 * billing period that collection closes. Prefer month label; fall back
 * to the period boundary inferred from the collection timestamp.
 */

function defaultExtendedDaysForCollection_(collectionDate, monthLabel) {
  const periodEnd = periodEndFromMonthLabel_(monthLabel) || getBillingPeriodStart_(collectionDate);
  const periodStart = new Date(
    periodEnd.getFullYear(),
    periodEnd.getMonth() - 1,
    periodEnd.getDate(),
    periodEnd.getHours(),
    periodEnd.getMinutes(),
    0,
    0
  );
  const days = (collectionDate.getTime() - periodStart.getTime()) / 86400000;
  if (isFinite(days) && days > 0) return Math.round(days);
  return Math.round(billingPeriodLengthDays_(periodEnd));
}

function standardDaysForCollection_(collection) {
  if (!collection) return 30;
  const fromLabel = periodEndFromMonthLabel_(collection.month);
  if (fromLabel) return billingPeriodLengthDays_(fromLabel);
  const timePart = collection.time && String(collection.time).length ? collection.time : "00:00";
  const dt = new Date(collection.date + "T" + timePart + ":00");
  if (isNaN(dt.getTime())) return 30;
  // Collection near/after the cycle point closes the period ending at
  // getBillingPeriodStart_(dt) when that start is on/before the collection.
  const boundary = getBillingPeriodStart_(dt);
  return billingPeriodLengthDays_(boundary);
}

function proRataForMeter_(baseline,present,extendedDays,standardDays){
  const std = standardDays != null && standardDays !== ""
    ? Number(standardDays)
    : billingPeriodLengthDays_(getBillingPeriodStart_(new Date()));
  return calculateProRata_(baseline,present,extendedDays,std);
}

  let byGraceDt = null;
  const GRACE_MS = 12 * 86400000;
  const graceEnd = new Date(boundaryDate.getTime() + GRACE_MS);
  const graceStart = new Date(boundaryDate.getTime() - 40 * 86400000);
  collections.forEach(function(c) {
    if (String(c.meter || "").trim() !== String(meter || "").trim()) return;
    const timePart = c.time && String(c.time).length ? c.time : "00:00";
    const dt = new Date(c.date + "T" + timePart + ":00");
    if (isNaN(dt.getTime())) return;
    const monthLabel = String(c.month || "").trim().toLowerCase();
    if (priorMonthLabel && monthLabel === priorMonthLabel) {
      if (!byLabelDt || dt > byLabelDt) {
        byLabelDt = dt;
        byLabel = c;
      }
    }
    if (dt >= graceStart && dt < graceEnd) {
      if (!byGraceDt || dt > byGraceDt) {
        byGraceDt = dt;
        byGrace = c;
      }
    }
  });
  return byLabel || byGrace;
}
function carryForwardForMeter_(meter, boundaryDate, standardDays, readings) {
  const c = getLatestPriorCollection_(meter, boundaryDate);
  if (!c) return null;
  const std = (standardDays != null && standardDays !== "" && Number(standardDays) > 0)
    ? Number(standardDays)
    : billingPeriodLengthDays_(boundaryDate);
  const fromCollection = standardDaysForCollection_(c);
  const useDays = fromCollection || std;

  // Billed units always from official collection raw reading.
  const billedProRata = calculateProRata_(
    c.previousBaseline, c.rawReading, c.extendedDays, useDays
  );
  if (!billedProRata) return null;

  // Carry-forward only: use live meter reading at the billing-period boundary
  // (start of this month / end of the period just closed), not the later raw.
  let boundaryReading = null;
  if (readings && readings.length) {
    const at = interpolatedReadingsAt_(readings, boundaryDate);
    boundaryReading = meter === "METER 1" ? at.newReading : at.oldReading;
  }

  let carryForward = billedProRata.carryForward;
  let carryPresent = billedProRata.present;
  let carrySource = "collection";
  if (boundaryReading !== null && boundaryReading !== "" && isFinite(Number(boundaryReading))) {
    carryPresent = Number(boundaryReading);
    // (reading at period start − baseline) − billed units
    // equivalent to: boundaryReading − adjustedPresent
    carryForward = carryPresent - billedProRata.adjustedPresent;
    carrySource = "boundary";
  }

  return {
    baseline: billedProRata.baseline,
    present: billedProRata.present,
    extendedDays: billedProRata.extendedDays,
    actualUnits: billedProRata.actualUnits,
    dailyAverage: billedProRata.dailyAverage,
    standardDays: billedProRata.standardDays,
    billedUnits: billedProRata.billedUnits,
    adjustedPresent: billedProRata.adjustedPresent,
    carryForward: carryForward,
    boundaryReading: boundaryReading,
    carryPresent: carryPresent,
    carrySource: carrySource
  };
}
function estimateReadingAt_(readings, target, meter) {
  let before=null, after=null;
  for(let i=0;i<readings.length;i++){
    const r=readings[i];
    if(r.datetime<=target && r[meter+'Reading']!==null) before=r;
    if(r.datetime>target && r[meter+'Reading']!==null){ after=r; break; }
  }
  if(!before) return null;
  if(before.datetime.getTime()===target.getTime()) return before[meter+'Reading'];
  if(!after) return before[meter+'Reading'];
  return profileWeightedInterpolate_(before,after,target,meter);
}
function last24HoursConsumption_(readings, endReading) {
  if(!endReading) return {newMeter:'',oldMeter:'',total:''};
  const end=endReading.datetime;
  const start=new Date(end.getTime()-24*60*60*1000);
  const newStart=estimateReadingAt_(readings,start,'new');
  const oldStart=estimateReadingAt_(readings,start,'old');
  const newEnd=endReading.newReading;
  const oldEnd=endReading.oldReading;
  const n=calculateDifference_(newStart,newEnd);
  const o=calculateDifference_(oldStart,oldEnd);
  return {newMeter:n,oldMeter:o,total:sumKnown_(n,o)};
}
function goalPaceMetrics_(currentUsage, billingStart, billingEnd, now) {
  const goal = Number(getGeneralSettings_().goalCombinedUnits || 100);
  const usage = Number(currentUsage || 0);
  const totalDays = (billingEnd.getTime() - billingStart.getTime()) / 86400000;
  const elapsedDays = Math.max(0, Math.min(totalDays, (now.getTime() - billingStart.getTime()) / 86400000));
  const remainingDays = Math.max(0, totalDays - elapsedDays);
  const remainingUnits = goal - usage;
  const dailyAverage = elapsedDays > 0 ? usage / elapsedDays : 0;
  const targetDailyAverage = totalDays > 0 ? goal / totalDays : 0;
  let requiredDailyAverage = '';
  if(remainingDays > 0 && remainingUnits > 0) {
    requiredDailyAverage = remainingUnits / remainingDays;
  }
  const paceDifference = dailyAverage - targetDailyAverage;
  return {
    goal:goal, elapsedDays:elapsedDays, remainingDays:remainingDays,
    remainingUnits:remainingUnits, dailyAverage:dailyAverage,
    targetDailyAverage:targetDailyAverage, requiredDailyAverage:requiredDailyAverage,
    paceDifference:paceDifference, achieved:usage <= goal && remainingDays === 0,
    overGoal:remainingUnits < 0
  };
}

function meterOperatingDays_(readings,start,end,meter,meterChange){
  if(end<=start) return 0;
  if(meterChange && meterChange.estimatedAt){
    const sw=meterChange.estimatedAt;
    const activeAfter = meterChange.activeAfter || "new";
    if(meter === activeAfter) return Math.max(0,(end-Math.max(start,sw))/86400000);
    return Math.max(0,(Math.min(end,sw)-start)/86400000);
  }
  return meterDaysCovered_(readings,start,end,meter);
}
function meterDaysCovered_(readings, start, end, meter) {
  const inputKey=meter==="new" ? "newInput" : "oldInput";
  const relevant=readings.filter(function(r){
    return r.datetime>=start && r.datetime<=end &&
      r[inputKey]!==null && r[inputKey]!=="" && isFinite(Number(r[inputKey]));
  });
  if(relevant.length<2){
    if(relevant.length===1){
      return Math.max(0,(end.getTime()-relevant[0].datetime.getTime())/86400000);
    }
    return 0;
  }
  return (relevant[relevant.length-1].datetime-relevant[0].datetime)/86400000;
}
function profileWeightAt_(datetime) {
  const profile = CONFIG.ESTIMATION_PROFILE_KWH_PER_HOUR;
  const hour = datetime.getHours();
  return Math.max(0.000001, Number(profile[hour] || 0));
}
function profileWeightBetween_(start, end) {
  if (!start || !end || end <= start) return 0;
  let total = 0;
  let cursor = new Date(start);
  while (cursor < end) {
    const nextHour = new Date(cursor);
    nextHour.setMinutes(0, 0, 0);
    nextHour.setHours(nextHour.getHours() + 1);
    const segmentEnd =
      nextHour < end ? nextHour : end;
    const minutes =
      (segmentEnd - cursor) / 60000;
    total += profileWeightAt_(cursor) * minutes;
    cursor = segmentEnd;
  }
  return total;
}
function profileWeightedInterpolate_(before, after, target, meter) {
  if (!before || !after) return null;
  const b = before[meter + "Reading"];
  const a = after[meter + "Reading"];
  if (b === null || a === null) return null;
  if (after.datetime <= before.datetime) return b;
  if (a < b) return null;
  const totalWeight =
    profileWeightBetween_(before.datetime, after.datetime);
  const targetWeight =
    profileWeightBetween_(before.datetime, target);
  if (totalWeight <= 0) {
    const fraction =
      (target - before.datetime) /
      (after.datetime - before.datetime);
    return b + (a - b) * fraction;
  }
  const fraction =
    Math.max(0, Math.min(1, targetWeight / totalWeight));
  return b + (a - b) * fraction;
}
function interpolatedReadingsAt_(readings, boundary) {
  let before = null;
  let after = null;
  for (let i = 0; i < readings.length; i++) {
    const r = readings[i];
    if (r.datetime <= boundary) {
      if (r.newReading !== null || r.oldReading !== null) {
        before = r;
      }
    } else {
      after = r;
      break;
    }
  }
  function interpolate(meter) {
    const b =
      before && before[meter + "Reading"] !== null
        ? before
        : null;
    const a =
      after && after[meter + "Reading"] !== null
        ? after
        : null;
    if (!b) return null;
    if (b.datetime.getTime() === boundary.getTime()) {
      return b[meter + "Reading"];
    }
    if (a && a.datetime.getTime() === boundary.getTime()) {
      return a[meter + "Reading"];
    }
    if (a && a.datetime > b.datetime) {
      return profileWeightedInterpolate_(
        b,
        a,
        boundary,
        meter
      );
    }
    return b[meter + "Reading"];
  }
  return {
    newReading: interpolate("new"),
    oldReading: interpolate("old")
  };
}
const PRIOR_SHRINKAGE_DAYS = 3;
function priorPeriodDailyAverages_(readings, billingStart) {
  if (!readings.length || readings[0].datetime >= billingStart) return null;
  const priorStart = new Date(
    billingStart.getFullYear(), billingStart.getMonth() - 1, billingStart.getDate(),
    billingStart.getHours(), billingStart.getMinutes(), 0, 0
  );
  const priorEnd = new Date(billingStart);
  const days = (priorEnd - priorStart) / 86400000;
  if (days <= 0) return null;
  const startR = interpolatedReadingsAt_(readings, priorStart);
  const endR = interpolatedReadingsAt_(readings, priorEnd);
  const newTotal = calculateDifference_(startR.newReading, endR.newReading);
  const oldTotal = calculateDifference_(startR.oldReading, endR.oldReading);
  return {
    newPerDay: newTotal === "" ? null : newTotal / days,
    oldPerDay: oldTotal === "" ? null : oldTotal / days
  };
}
function blendedDailyRate_(observedTotal, observedDays, priorPerDay) {
  if (priorPerDay === null || typeof priorPerDay === "undefined") {
    return observedDays > 0 ? observedTotal / observedDays : 0;
  }
  return (observedTotal + priorPerDay * PRIOR_SHRINKAGE_DAYS) /
         (observedDays + PRIOR_SHRINKAGE_DAYS);
}
function projectBillingMonthFromDaily_(daily, billingStart, billingEnd, now, readings) {
  const startKey = Utilities.formatDate(
    billingStart,
    Session.getScriptTimeZone(),
    "yyyy-MM-dd"
  );
  const endKey = Utilities.formatDate(
    billingEnd,
    Session.getScriptTimeZone(),
    "yyyy-MM-dd"
  );
  const periodDaily = daily.filter(function(d) {
    return d.date >= startKey && d.date < endKey;
  });
  let completedNew = 0;
  let completedOld = 0;
  let completedDays = 0;
  let currentDay = null;
  periodDaily.forEach(function(d) {
    if (d.completed) {
      if (d.newMeter !== "") completedNew += Number(d.newMeter || 0);
      if (d.oldMeter !== "") completedOld += Number(d.oldMeter || 0);
      completedDays++;
    } else {
      currentDay = d;
    }
  });
  const fullPeriodDays = (billingEnd - billingStart) / 86400000;
  let partialFullNew = 0;
  let partialFullOld = 0;
  if (currentDay) {
    const dayStart = getFivePmDayStart_(now);
    const dayEnd = new Date(dayStart);
    dayEnd.setDate(dayEnd.getDate() + 1);
    const elapsedWeight = profileWeightBetween_(dayStart, now);
    const fullWeight = profileWeightBetween_(dayStart, dayEnd);
    const fraction = fullWeight > 0 ? elapsedWeight / fullWeight : 0;
    const partialNew = currentDay.newMeter === "" ? 0 : Number(currentDay.newMeter || 0);
    const partialOld = currentDay.oldMeter === "" ? 0 : Number(currentDay.oldMeter || 0);
    const completedDailyNew = [];
    const completedDailyOld = [];
    periodDaily.forEach(function(d) {
      if (!d.completed) return;
      if (d.newMeter !== "" && isFinite(Number(d.newMeter))) completedDailyNew.push(Number(d.newMeter));
      if (d.oldMeter !== "" && isFinite(Number(d.oldMeter))) completedDailyOld.push(Number(d.oldMeter));
    });
    const fallbackNew = completedDailyNew.length
      ? completedDailyNew.reduce(function(a,b){return a+b;},0) / completedDailyNew.length
      : 0;
    const fallbackOld = completedDailyOld.length
      ? completedDailyOld.reduce(function(a,b){return a+b;},0) / completedDailyOld.length
      : 0;
    if (fraction >= 0.10 && elapsedWeight > 0 && fullWeight > 0) {
      partialFullNew = partialNew * fullWeight / elapsedWeight;
      partialFullOld = partialOld * fullWeight / elapsedWeight;
    } else {
      partialFullNew = Math.max(partialNew, fallbackNew);
      partialFullOld = Math.max(partialOld, fallbackOld);
    }
  }
  const futureDays = Math.max(
    0,
    fullPeriodDays - completedDays - (currentDay ? 1 : 0)
  );
  let futureNewPerDay = 0;
  let futureOldPerDay = 0;
  const prior = priorPeriodDailyAverages_(readings, billingStart);
  const observedDaysCount = completedDays + (currentDay ? 1 : 0);
  const observedNewTotal = completedNew + partialFullNew;
  const observedOldTotal = completedOld + partialFullOld;
  const meterChange = readings ? estimateMeterChangeEvent_(readings, billingStart, billingEnd) : null;
  if (meterChange) {
    const sw = meterChange.estimatedAt;
    const activeAfter = meterChange.activeAfter || "new";
    const daysSinceSwap = Math.max(0, (now - Math.max(billingStart, sw)) / 86400000);
    const daysBeforeSwap = Math.max(0, (Math.min(now, sw) - billingStart) / 86400000);
    const meter1DaysCompleted = activeAfter === "new" ? daysSinceSwap : daysBeforeSwap;
    const meter2DaysCompleted = activeAfter === "old" ? daysSinceSwap : daysBeforeSwap;
    if (now >= sw) {
      if (activeAfter === "new") {
        futureNewPerDay = blendedDailyRate_(completedNew + partialFullNew, meter1DaysCompleted, prior && prior.newPerDay);
        futureOldPerDay = 0;
      } else {
        futureNewPerDay = 0;
        futureOldPerDay = blendedDailyRate_(completedOld + partialFullOld, meter2DaysCompleted, prior && prior.oldPerDay);
      }
    } else {
      if (activeAfter === "new") {
        futureNewPerDay = 0;
        futureOldPerDay = blendedDailyRate_(completedOld + partialFullOld, meter2DaysCompleted, prior && prior.oldPerDay);
      } else {
        futureNewPerDay = blendedDailyRate_(completedNew + partialFullNew, meter1DaysCompleted, prior && prior.newPerDay);
        futureOldPerDay = 0;
      }
    }
  } else {
    futureNewPerDay = blendedDailyRate_(observedNewTotal, observedDaysCount, prior && prior.newPerDay);
    futureOldPerDay = blendedDailyRate_(observedOldTotal, observedDaysCount, prior && prior.oldPerDay);
  }
  const projectedNew =
    completedNew + partialFullNew + futureNewPerDay * futureDays;
  const projectedOld =
    completedOld + partialFullOld + futureOldPerDay * futureDays;
  return {
    newMeter: projectedNew,
    oldMeter: projectedOld,
    total: projectedNew + projectedOld
  };
}
function getInitialHourlyData_(readings,dayKey){
  if(!dayKey) return [];
  const dayStart=new Date(dayKey+"T17:00:00");
  const dayEnd=new Date(dayStart);
  dayEnd.setDate(dayEnd.getDate()+1);
  const now=new Date();
  let effectiveEnd=dayEnd;
  if(now>=dayStart && now<dayEnd){
    effectiveEnd=now;
  }else if(now<dayStart){
    return [];
  }
  return getHourlyChartData_(readings,dayStart,effectiveEnd);
}
function getHourlyChartData_(readings, dayStart, dayEnd) {
  const output = [];
  let cursor = new Date(dayStart);
  while (cursor < dayEnd) {
    const next = new Date(cursor);
    next.setHours(next.getHours() + 1);
    const startR = estimateReadingAt_(readings, cursor, 'new');
    const startO = estimateReadingAt_(readings, cursor, 'old');
    const endR = estimateReadingAt_(readings, next, 'new');
    const endO = estimateReadingAt_(readings, next, 'old');
    const newMeter = calculateDifference_(startR, endR);
    const oldMeter = calculateDifference_(startO, endO);
    const total = sumKnown_(newMeter, oldMeter);
    output.push({
      label: Utilities.formatDate(cursor, Session.getScriptTimeZone(), "h a"),
      start: cursor.getTime(),
      end: next.getTime(),
      newMeter: newMeter,
      oldMeter: oldMeter,
      total: total
    });
    cursor = next;
  }
  return output;
}
function getDailyChartData_(readings, periodStart, periodEnd, calculationNow) {
  if (!readings.length) return [];
  const output=[];
  let day=new Date(periodStart);
  const now=calculationNow || new Date();
  while(day < periodEnd){
    const start=new Date(day);
    const end=new Date(day);
    end.setDate(end.getDate()+1);
    const startR=interpolatedReadingsAt_(readings,start);
    const completed=now >= end;
    let endR;
    let estimatedEndR=false;
    let profileFraction=1;
    if(completed){
      endR=interpolatedReadingsAt_(readings,end);
    }else{
      const actualEndR=readingsAtOrBefore_(readings,now);
      endR={newReading:null,oldReading:null};
      const elapsedWeight=profileWeightBetween_(start,now);
      const fullWeight=profileWeightBetween_(start,end);
      if(elapsedWeight>0&&fullWeight>0&&startR.newReading!==null&&actualEndR.newReading!==null){
        const observed=calculateDifference_(startR.newReading,actualEndR.newReading);
        if(observed!==""){endR.newReading=Number(startR.newReading)+Number(observed)*fullWeight/elapsedWeight;estimatedEndR=true;}
      }
      if(elapsedWeight>0&&fullWeight>0&&startR.oldReading!==null&&actualEndR.oldReading!==null){
        const observed=calculateDifference_(startR.oldReading,actualEndR.oldReading);
        if(observed!==""){endR.oldReading=Number(startR.oldReading)+Number(observed)*fullWeight/elapsedWeight;estimatedEndR=true;}
      }
      profileFraction=elapsedWeight>0&&fullWeight>0?elapsedWeight/fullWeight:0;
    }
    const newMeter=calculateDifference_(startR.newReading,endR.newReading);
    const oldMeter=calculateDifference_(startR.oldReading,endR.oldReading);
    const total=sumKnown_(newMeter,oldMeter);
    if(endR.newReading!==null||endR.oldReading!==null){
      output.push({
        label:Utilities.formatDate(start,Session.getScriptTimeZone(),"dd"),
        date:Utilities.formatDate(start,Session.getScriptTimeZone(),"yyyy-MM-dd"),
        period:Utilities.formatDate(start,Session.getScriptTimeZone(),"dd MMM, '5:00 PM'")+" → "+Utilities.formatDate(end,Session.getScriptTimeZone(),"dd MMM, '5:00 PM'"),
        newMeter:newMeter,oldMeter:oldMeter,total:total,completed:completed,estimated:estimatedEndR,profileFraction:profileFraction,endTime:end.getTime()
      });
    }
    day.setDate(day.getDate()+1);
  }
  return output;
}

function getRecordsForClient_() {
  const sheet = findResponseSheet_(getSpreadsheet_());
  if (!sheet) return [];

  if (values.length <= 1) return [];
  const headers = values[0];

    if (row[idx.date] === "" || row[idx.time] === "") continue;

      rowNumber: i + 1,

      time: Utilities.formatDate(
        dt,
        Session.getScriptTimeZone(),
        "HH:mm"
      ),
      newMeter:
        idx.newMeter >= 0 && row[idx.newMeter] !== ""
          ? row[idx.newMeter]
          : "",
      oldMeter:
        idx.oldMeter >= 0 && row[idx.oldMeter] !== ""
          ? row[idx.oldMeter]
          : "",
      loadKw:
        idx.loadKw >= 0 && row[idx.loadKw] !== "" ? row[idx.loadKw] :
        (idx.notes >= 0 && row[idx.notes] !== "" ? row[idx.notes] : ""),
      notes:
        idx.notes >= 0 ? row[idx.notes] : "",
      sortKey: dt.getTime()
    });
  }
  records.sort(function(a, b) {
    return b.sortKey - a.sortKey;
  });
  records.forEach(function(r) {
    delete r.sortKey;
  });
  return records;
}
function saveRecord(record) {
  if(record && record.rowNumber) requireAppOwner_();
  const sheet = findResponseSheet_(getSpreadsheet_());
  if (!sheet) {
    throw new Error("No response sheet found.");
  }
  const headers = sheet
    .getRange(1, 1, 1, sheet.getLastColumn())
    .getValues()[0];
  const idx = getColumnIndices_(headers);
  if (idx.date < 0 || idx.time < 0) {
    throw new Error("Required columns not found.");
  }
  if (!record || !record.date || !record.time) {
    throw new Error("Date and time are required.");
  }
  const dateParts = String(record.date).split("-");
  const timeParts = String(record.time).split(":");
  const dt = new Date(
    Number(dateParts[0]),
    Number(dateParts[1]) - 1,
    Number(dateParts[2]),
    Number(timeParts[0]),
    Number(timeParts[1]),
    0,
    0
  );
  if (isNaN(dt.getTime())) {
    throw new Error("Invalid date or time.");
  }
  const newMeter =
    record.newMeter === "" ||
    record.newMeter === null ||
    typeof record.newMeter === "undefined"
      ? ""
      : Number(record.newMeter);
  const oldMeter =
    record.oldMeter === "" ||
    record.oldMeter === null ||
    typeof record.oldMeter === "undefined"
      ? ""
      : Number(record.oldMeter);
  if (newMeter !== "" && isNaN(newMeter)) {
    throw new Error("Invalid Meter 1 reading.");
  }
  if (oldMeter !== "" && isNaN(oldMeter)) {
    throw new Error("Invalid Meter 2 reading.");
  }
  const targetRow =
    record.rowNumber
      ? Number(record.rowNumber)
      : sheet.getLastRow() + 1;
  sheet.getRange(targetRow, idx.date + 1).setValue(dt);
  sheet.getRange(targetRow, idx.time + 1).setValue(dt);
  if (idx.newMeter >= 0) {
    sheet.getRange(targetRow, idx.newMeter + 1).setValue(newMeter);
  }
  if (idx.oldMeter >= 0) {
    sheet.getRange(targetRow, idx.oldMeter + 1).setValue(oldMeter);
  }
  const loadValue=record.loadKw===""||record.loadKw===null||typeof record.loadKw==="undefined" ? "" : Number(record.loadKw);
  if(loadValue!=="" && isNaN(loadValue)) throw new Error("Invalid inverter reading.");
  const loadColumn=idx.loadKw>=0 ? idx.loadKw : idx.notes;
  if(loadColumn>=0) sheet.getRange(targetRow,loadColumn+1).setValue(loadValue);
  rebuildData();
  return {success: true};
}

function combineDateTime_(dateValue, timeValue) {
  const d = dateValue instanceof Date
    ? new Date(dateValue)
    : new Date(dateValue);
  if (isNaN(d.getTime())) return null;
  let hours = 0;
  let minutes = 0;
  let seconds = 0;
  if (timeValue instanceof Date) {
    hours = timeValue.getHours();
    minutes = timeValue.getMinutes();
    seconds = timeValue.getSeconds();
  } else {
    const text = String(timeValue).trim().toUpperCase();
    const match = text.match(
      /(\d{1,2}):(\d{2})(?::(\d{2}))?\s*(AM|PM)?/
    );
    if (!match) return null;
    hours = Number(match[1]);
    minutes = Number(match[2]);
    seconds = match[3] ? Number(match[3]) : 0;
    const ampm = match[4];
    if (ampm === "PM" && hours < 12) hours += 12;
    if (ampm === "AM" && hours === 12) hours = 0;
  }
  d.setHours(hours, minutes, seconds, 0);
  return d;
}

function collectionDateTime_(c) {
  const timePart = c.time && String(c.time).length ? c.time : "00:00";
  const dt = new Date(String(c.date) + "T" + timePart + ":00");
  return isNaN(dt.getTime()) ? null : dt;
}
function getReadingCollections_() {
  const ss = getSpreadsheet_();
  const sheet = ensureReadingCollectionsSheet_(ss);
  const values = sheet.getDataRange().getValues();
  const result = [];
  for (let i = 1; i < values.length; i++) {
    if (!values[i][0]) continue;
    const rawDate = values[i][2];
    const dateStr = rawDate instanceof Date
      ? Utilities.formatDate(rawDate, Session.getScriptTimeZone(), "yyyy-MM-dd")
      : String(rawDate || "");
    const timeStr = safeTimeLabel_(values[i][3]);
    result.push({
      rowNumber: i + 1,
      month: safeMonthLabel_(values[i][0]),
      meter: String(values[i][1]),
      date: dateStr,
      time: timeStr,
      previousBaseline: Number(values[i][4] || 0),
      rawReading: Number(values[i][5] || 0)
    });
  }
  // Extended days are always calculated from timestamps — never read from the sheet.
  const byMeter = {};
  result.forEach(function(c) {
    const key = String(c.meter || "");
    if (!byMeter[key]) byMeter[key] = [];
    byMeter[key].push(c);
  });
  Object.keys(byMeter).forEach(function(key) {
    const list = byMeter[key].slice().sort(function(a, b) {
      const da = collectionDateTime_(a);
      const db = collectionDateTime_(b);
      return (da ? da.getTime() : 0) - (db ? db.getTime() : 0);
    });
    for (let i = 0; i < list.length; i++) {
      const c = list[i];
      const dt = collectionDateTime_(c);
      c.standardDays = standardDaysForCollection_(c);
      if (i > 0) {
        const prevDt = collectionDateTime_(list[i - 1]);
        c.extendedDays = Math.round(
          (dt && prevDt)
            ? (dt.getTime() - prevDt.getTime()) / 86400000
            : defaultExtendedDaysForCollection_(dt || new Date(), c.month)
        );
      } else {
        c.extendedDays = Math.round(defaultExtendedDaysForCollection_(dt || new Date(), c.month));
      }
    }
  });
  return result;
}
function addReadingCollection(collection) {
  requireAppOwner_();
  if (!collection || !collection.meter || !collection.date || !collection.time ||
      collection.rawReading === "" || collection.rawReading === null || typeof collection.rawReading === "undefined") {
    throw new Error("Meter, collection date, collection time and raw reading are required.");
  }
  const meter = String(collection.meter).trim();
  const dateStr = String(collection.date).trim();
  const timeStr = String(collection.time).trim();
  const rawReading = Number(collection.rawReading);
  if (isNaN(rawReading)) throw new Error("Invalid raw reading.");
  const dateParts = dateStr.split("-");
  const timeParts = timeStr.split(":");
  const dt = new Date(
    Number(dateParts[0]),
    Number(dateParts[1]) - 1,
    Number(dateParts[2]),
    Number(timeParts[0] || 0),
    Number(timeParts[1] || 0),
    0,
    0
  );
  if (isNaN(dt.getTime())) {
    throw new Error("Invalid collection date or time.");
  }
  const hasOverrideMonth = collection.month !== "" && collection.month !== null && typeof collection.month !== "undefined";
  const month = hasOverrideMonth
    ? String(collection.month).trim()
    : formatBillingMonth_(addMonth_(getBillingPeriodStart_(dt)));
  const previous = getLatestPriorCollection_(meter, dt);
  const hasOverrideBaseline = collection.previousBaseline !== "" && collection.previousBaseline !== null &&
    typeof collection.previousBaseline !== "undefined" && !isNaN(Number(collection.previousBaseline));
  const previousBaseline = hasOverrideBaseline
    ? Number(collection.previousBaseline)
    : (previous ? previous.rawReading : 0);
  // extendedDays is not stored; computed on read from collection timestamps
  const ss = getSpreadsheet_();
  const sheet = ensureReadingCollectionsSheet_(ss);
  const values = sheet.getDataRange().getValues();
  let targetRow = values.length + 1;
  for (let i = 1; i < values.length; i++) {
    if (String(values[i][0]).trim().toLowerCase() === month.toLowerCase() &&
        String(values[i][1]).trim().toLowerCase() === meter.toLowerCase()) {
      targetRow = i + 1;
      break;
    }
  }
  sheet.getRange(targetRow, 1, 1, 1).setNumberFormat("@");
  sheet.getRange(targetRow, 4, 1, 1).setNumberFormat("@");
  sheet.getRange(targetRow, 1, 1, 6).setValues([[
    month, meter, dt, timeStr, previousBaseline, rawReading
  ]]);
  rebuildData();
  const computedExtended = previous
    ? (function() {
        const prevTimePart = previous.time && String(previous.time).length ? previous.time : "00:00";
        const prevDt = new Date(previous.date + "T" + prevTimePart + ":00");
        return isNaN(prevDt.getTime())
          ? defaultExtendedDaysForCollection_(dt, month)
          : (dt - prevDt) / 86400000;
      })()
    : defaultExtendedDaysForCollection_(dt, month);
  return {success: true, month: month, previousBaseline: previousBaseline, extendedDays: computedExtended};
}

function isAuthExpiryError_(error){
  const msg = (error && error.message) ? String(error.message) : String(error || "");
  return /permission to access the requested document/i.test(msg) ||
         /authorization is required/i.test(msg);
}
function handleAuthExpiry_(error){
  if(!isAuthExpiryError_(error)) return false;
  if(confirm("Your session has expired (this page was open a while). Reload to continue?")){
    window.location.reload();
  }
  return true;
}

function renderTariff(settings){

    {title:"Energy rates", keys:["0-100 Rate","100-200 Rate","201-300 Rate","301-400 Rate","401-500 Rate","501-600 Rate","601-700 Rate","Above 700 Rate"]},
    {title:"Fixed charges", keys:["0-100 Fixed","100-200 Fixed","201-300 Fixed","301-400 Fixed","401-500 Fixed","501-600 Fixed","601-700 Fixed","Above 700 Fixed"]},

  ];
  let html='<div class="tariffIntro">Edit the values below only when your tariff or applicable bill adjustments change.</div>';
  groups.forEach(function(group){
    const rows=settings.rows.filter(function(r){return group.keys.indexOf(r.key)!==-1;});
    if(!rows.length) return;
    html+='<div class="tariffGroup"><div class="tariffGroupTitle">'+escapeHtml(group.title)+'</div><div class="tariffGrid">';
    rows.forEach(function(r){html+=tariffInput_(r);});

  const known={};
  groups.forEach(function(g){g.keys.forEach(function(k){known[k]=true;});});
  const other=settings.rows.filter(function(r){return !known[r.key];});
  if(other.length){
    html+='<div class="tariffGroup"><div class="tariffGroupTitle">Other parameters</div><div class="tariffGrid">';
    other.forEach(function(r){html+=tariffInput_(r);});
    html+='</div></div>';

  document.getElementById("tariffContent").innerHTML=html;

    data.dashboard.availableBillingMonths = months;

  }

function num(value){

function money(value){
  if(value === "" || value === null || typeof value === "undefined"){
    return "—";
  }
  return "Rs " + Number(value).toLocaleString("en-PK",{
    minimumFractionDigits:0,
    maximumFractionDigits:0
  });
}
function formatRelativeTime(timestamp){
  if(!timestamp) return "";
  const diffMs = Date.now() - timestamp;
  const diffMin = Math.round(diffMs / 60000);
  if(diffMin < 1) return "just now";
  if(diffMin < 60) return diffMin + " min ago";
  const diffHours = Math.round(diffMs / 3600000);
  if(diffHours < 72){
    return diffHours + (diffHours === 1 ? " hour ago" : " hours ago");
  }
  const diffDays = Math.round(diffMs / 86400000);
  return diffDays + (diffDays === 1 ? " day ago" : " days ago");
}
function escapeHtml(value){
  if(value === null || typeof value === "undefined") return "";
  return String(value)
    .replace(/&/g,"&amp;")
    .replace(/"/g,"&quot;")
    .replace(/</g,"&lt;")
    .replace(/>/g,"&gt;");
}

function renderHourlyChart(hourly, selectedDay){
  let html = '<div class="section"><div class="chart">';
  html += '<div class="hourlyHeader">';
  html += '<div class="label">Hourly Consumption</div>';
  html += '<select class="hourlyDaySelect" onchange="changeHourlyDay(this.value)">';
 (function() {

    if (days.length > 1) {
        days.push(days.shift());

    days.reverse().forEach(function(d){
        html += '<option value="' + escapeHtml(d.value) + '"' + (d.value === selectedDay ? ' selected' : '') + '>' + escapeHtml(d.label) + '</option>';
    });
})();
  html += '</select></div>';
  let maximum = 0;
  hourly.forEach(function(d){
    if(d.total !== "" && d.total !== null) maximum = Math.max(maximum, Number(d.total));
  });
  if(maximum <= 0) maximum = 1;
  const chartMax = niceHourlyMaximum_(maximum);
  const ySteps = chartMax <= 0.6 ? 3 : (chartMax <= 1 ? 4 : 5);
  html += '<div class="graph hourlyGraph">';
  html += '<div class="yAxis">';
  for(let i=ySteps;i>=0;i--){
    html += '<div>' + formatAxisNumber(chartMax * i / ySteps) + '</div>';
  }
  html += '</div><div class="plotArea"><div class="graphBars hourlyBars">';
  hourly.forEach(function(d){

    const b = d.oldMeter === "" ? 0 : Number(d.oldMeter);

      html += '<div class="graphBar hourlyBar" style="height:' + height + '%" title="' +
        escapeHtml(d.label) + ': ' + num(total) + ' kWh">';
      const bPct = total > 0 ? (b / total * 100) : 0;
      const aPct = total > 0 ? (a / total * 100) : 0;
      if(b > 0) html += '<div class="graphSegment graphSegment2" style="flex:' + bPct + '"></div>';
      if(a > 0) html += '<div class="graphSegment graphSegment1" style="flex:' + aPct + '"></div>';
      html += '<div class="graphValue">' + num(total) + '</div>';
      html += '</div>';
    }
    html += '<div class="graphLabel">' + escapeHtml(d.label) + '</div>';
    html += '</div>';
  });
  html += '</div></div></div></div>';
  return html;

function niceChartMaximum(value){
  value=Number(value)||0;
  if(value<=0) return 2;
  if(value<=10) return Math.ceil(value/2)*2;
  return Math.ceil(value/2)*2;
}
function niceHourlyMaximum_(value){
  value=Number(value)||0;
  if(value<=0) return 0.6;
  const target=Math.max(0.6,value*1.08);
  const magnitude=Math.pow(10,Math.floor(Math.log10(target)));
  const normalized=target/magnitude;
  let step;
  if(normalized<=1) step=1;
  else if(normalized<=2) step=2;
  else if(normalized<=5) step=5;
  else step=10;
  return step*magnitude;
}
function formatAxisNumber(value){
  if(value >= 100) return Math.round(value);
  if(value >= 10) return Number(value).toFixed(0);
  return Number(value).toFixed(value%1===0 ? 0 : 1);
}

  if(split && split.meter1 && split.meter2){
    p1=split.meter1; p2=split.meter2; total=split.total; units=split.units;

  document.getElementById("billContent").innerHTML='<div class="billInfo"><div class="loading">Calculating 50/50 split...</div></div>';

function proRataBlock_(data){
  const p=data && data.proRata;
  if(!p) return '';
  const rows=[
    ['Previous billed baseline','baseline','kWh'],
    ['Present live reading','present','kWh'],
    ['Total actual units','actualUnits','kWh'],
    ['Extended interval','extendedDays','days'],
    ['Daily velocity average','dailyAverage','kWh/d'],
    ['Standard cycle','standardDays','days'],
    ['Normalized billed units','billedUnits','kWh'],
    ['Adjusted present reading','adjustedPresent','kWh'],
    ['Carry-forward balance','carryForward','kWh']
  ];
  function meter(title,x){
    if(!x) return '<div class="proRataMeter"><h3>'+title+'</h3><div class="panelIntro">Insufficient baseline/present reading data for this meter.</div></div>';
    let h='<div class="proRataMeter"><h3>'+title+'</h3><div class="proRataRows">';
    rows.forEach(function(r){
      const v=x[r[1]];
      h+='<div><span>'+escapeHtml(r[0])+'</span><strong>'+num(v)+' <span class="unit">'+r[2]+'</span></strong></div>';
    });
    h+='</div></div>'; return h;
  }
  return '<details class="proRataDetails"><summary>Pro-Rata normalization</summary>'+
    '<div class="tariffIntro">Applies the supplied deterministic formula: actual units ÷ extended days × standard cycle, rounded down to whole billed units. This is an audit view and does not replace the existing projected-bill calculation.</div>'+
    meter('METER 1',p.meter1)+meter('METER 2',p.meter2)+
    '<div class="proRataFormula">Billed Units = floor((Present Reading − Previous Billed Baseline) ÷ Extended Days × Standard Days)</div>'+
    '</details>';
}

      '<td><input disabled type="number" step="0.01" id="oldMeter-' + rn +

      '<td><input disabled type="number" step="0.01" id="loadKw-' + rn + '" value="' + escapeHtml(r.loadKw || "") + '"></td>' +

    '</tr>';
  });

  if(!records.length){
    html += '<div style="padding:20px;color:#6b7280;text-align:center">' +
      'No readings yet — add the first one above.' +
      '</div>';
  }
  document.getElementById("recordsContent").innerHTML = html;
  const now = new Date();
  const pad = n => n < 10 ? "0" + n : String(n);
  document.getElementById("new-date").value =
    now.getFullYear() + "-" +
    pad(now.getMonth()+1) + "-" +
    pad(now.getDate());
  document.getElementById("new-time").value =
    pad(now.getHours()) + ":" + pad(now.getMinutes());
  if(records.length){
    const latest = records[0];
    const nm = document.getElementById("new-newMeter");
    const om = document.getElementById("new-oldMeter");
    if(nm && latest.newMeter !== "" && latest.newMeter !== null) nm.value = latest.newMeter;
    if(om && latest.oldMeter !== "" && latest.oldMeter !== null) om.value = latest.oldMeter;
  }
}

function renderHistory(billingHistory, readingCollections){
  const host = document.getElementById('billingHistoryContent');
  if(host){
    const meter1Rows = billingHistory.filter(function(r){ return r.meter === 'METER 1'; });
    const meter2Rows = billingHistory.filter(function(r){ return r.meter === 'METER 2'; });
    host.innerHTML = historyTable_('Meter 1', meter1Rows) + historyTable_('Meter 2', meter2Rows);
  }
  const auditHost = document.getElementById('readingCollectionsContent');
  if(auditHost){
    if(!readingCollections.length){
      auditHost.innerHTML = '<div style="padding:20px;color:#6b7280;text-align:center">No collections logged yet.</div>';
    } else {
      let html = '';
      readingCollections.forEach(function(c){
        const actualUnits = c.rawReading - c.previousBaseline;
        const extendedDays = Math.round(Number(c.extendedDays || 0));
        const dailyVelocity = extendedDays > 0 ? actualUnits / extendedDays : 0;
        // Each collection uses that billing month's real length (30 or 31, etc.)
        const standardDays = Math.round(
          (c.standardDays != null && c.standardDays !== '')
            ? Number(c.standardDays)
            : ((lastDashboardData && lastDashboardData.proRata && lastDashboardData.proRata.standardDays) || 30)
        );
        const billedUnits = Math.floor(dailyVelocity * standardDays);
        const adjustedPresent = c.previousBaseline + billedUnits;
        const carryForward = actualUnits - billedUnits;
        const daysLabel = Number.isInteger(standardDays) ? String(standardDays) : Number(standardDays).toFixed(0);
        html += '<div class="auditCard"><h4>' + escapeHtml(c.month) + ' — ' + escapeHtml(c.meter) + '</h4>' +
          '<div class="auditRow"><span>Collected</span><strong>' + escapeHtml(c.date) + ' ' + escapeHtml(c.time) + '</strong></div>' +
          '<div class="auditRow"><span>Previous baseline</span><strong>' + num(c.previousBaseline) + ' kWh</strong></div>' +
          '<div class="auditRow"><span>Raw reading</span><strong>' + num(c.rawReading) + ' kWh</strong></div>' +
          '<div class="auditRow"><span>Extended days</span><strong>' + extendedDays + '</strong></div>' +
          '<div class="auditRow"><span>Total actual units</span><strong>' + num(actualUnits) + ' kWh</strong></div>' +
          '<div class="auditRow"><span>Daily velocity</span><strong>' + num(dailyVelocity) + ' kWh/d</strong></div>' +
          '<div class="auditRow"><span>Billed units (' + daysLabel + 'd)</span><strong>' + num(billedUnits) + ' kWh</strong></div>' +
          '<div class="auditRow"><span>Adjusted present counter</span><strong>' + num(adjustedPresent) + ' kWh</strong></div>' +
          '<div class="auditRow"><span>Carry-forward balance</span><strong>' + num(carryForward) + ' kWh</strong></div>' +
          '</div>';
      });
      auditHost.innerHTML = html;
    }
  }
}

    });
